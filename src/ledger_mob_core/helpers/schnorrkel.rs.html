1//! Schrnorrkel / fog authority signing helpers
2//!
3//! These implement the same behaviour as the upstream `Signer` and `Verifier`
4//! traits, with modifications to reduce stack use for execution on the ledger.
5//!

18// Create a deterministic nonce using a merlin transcript. See this crate's
19// README for a security statement.
20#[inline(never)]
21fn schnorrkel_nonce(private_key: &RistrettoPrivate, context: &[u8], message: &[u8]) -> [u8; 32] {
22    let mut transcript = MerlinTranscript::new(b"SigningNonce");
23    transcript.append_message(b"context", context);
24    transcript.append_message(b"private", private_key.as_ref());
25    transcript.append_message(b"message", message);
26    let mut nonce = [0u8; 32];
27    transcript.challenge_bytes(b"nonce", &mut nonce);
28    nonce
29}

31// Construct a Schnorrkel SecretKey object from ourselves, and our nonce value
32#[inline(never)]
33fn schnorrkel_secret(private_key: &RistrettoPrivate, nonce: &[u8; 32]) -> SchnorrkelPrivate {
34    let mut secret_bytes = [0u8; 64];
35    secret_bytes[0..32].copy_from_slice(private_key.as_ref());
36    secret_bytes[32..64].copy_from_slice(&nonce[..]);
37    SchnorrkelPrivate::from_bytes(&secret_bytes).unwrap()
38}

40#[inline(never)]
41fn schnorrkel_sign(
42    private_key: &RistrettoPrivate,
43    context: &[u8],
44    message: &[u8],
45) -> RistrettoSignature {
46    // Create a deterministic nonce using a merlin transcript. See this crate's
47    // README for a security statement.
48    let nonce = schnorrkel_nonce(private_key, context, message);
49
50    // Construct a Schnorrkel SecretKey object from ourselves, and our nonce value
51    let secret_key = schnorrkel_secret(private_key, &nonce);
52    let keypair = secret_key.to_keypair();
53
54    // SigningContext provides domain separation for signature
55    let mut t = MerlinTranscript::new(b"SigningContext");
56    t.append_message(b"", context);
57    t.append_message(b"sign-bytes", message);
58    // NOTE: This signature is deterministic due to using the above nonce as the rng
59    // seed
60
61    // Setup HC128 RNG core
62    let mut core = Hc128Core::from_seed(nonce);
63
64    // Wrap this in BlockRng
65    // using a pointer container to avoid further stack allocation
66    let container = Hc128CoreContainer(&mut core);
67    let mut csprng = BlockRng::new(container);
68
69    let mut transcript = attach_rng(t, &mut csprng);
70    RistrettoSignature::from(keypair.sign(&mut transcript))
71}

73/// Wrapper allowing [Hc128Core] reference to implement [BlockRngCore],
74///  working around the lack of a blanket [BlockRngCore] impl
75/// for `&mut T` where `T: BlockRngCore`.
76struct Hc128CoreContainer<'a>(&'a mut Hc128Core);
77
78impl<'a> BlockRngCore for Hc128CoreContainer<'a> {
79    type Item = u32;
80    type Results = [u32; 16];
81
82    fn generate(&mut self, results: &mut Self::Results) {
83        self.0.generate(results);
84    }
85}
86
87impl<'a> rand_core::CryptoRng for Hc128CoreContainer<'a> {}

89/// Canonical signing context byte string
90const CONTEXT: &[u8] = b"Fog authority signature";
91
92/// Re-implementation of `Signer` for `RistrettoPrivate` using reference types
93/// to minimize stack allocations
94#[inline(never)]
95pub fn sign_authority(
96    private_key: &SubaddressViewPrivate,
97    spki_bytes: &[u8],
98) -> RistrettoSignature {
99    schnorrkel_sign(private_key.as_ref(), CONTEXT, spki_bytes)
100}

102#[cfg(test)]
103mod test {
104    use mc_account_keys::{AccountKey, DEFAULT_SUBADDRESS_INDEX};
105    use mc_crypto_keys::RistrettoPublic;
106    use mc_fog_sig_authority::{Signer, Verifier};

115    const FOGS: &[FogId] = &[
116        FogId::MobMain,
117        FogId::MobTest,
118        FogId::SignalMain,
119        FogId::SignalTest,
120    ];
121
122    #[test]
123    fn schnorrkel_sign_verify() {
124        for _i in 0..10 {
125            let k = RistrettoPrivate::from_random(&mut OsRng {});
126            let p = RistrettoPublic::from(&k);
127            let spki = FogId::MobMain.spki();
128
129            // Generate using upstream impl
130            let s1 = k.sign_authority(spki).unwrap();
131            p.verify_authority(spki, &s1).unwrap();
132
133            // Generate using local impl
134            let s2 = sign_authority(&k.into(), spki);
135            p.verify_authority(spki, &s2).unwrap();
136
137            // Check impls match
138            assert_eq!(s1, s2);
139        }
140    }
141
142    #[test]
143    fn fog_authority_sigs() {
144        for f in FOGS {
145            for _i in 0..10 {
146                // Create random account with fog info
147                let spki = f.spki();
148                let a = AccountKey::random(&mut OsRng {}).with_fog(f.url(), "", spki);
149
150                // Using the default subaddress
151                let subaddr = a.subaddress(DEFAULT_SUBADDRESS_INDEX);
152                let view_private = a.default_subaddress_view_private();
153
154                // Generate authority signature using local impl
155                let sig: [u8; 64] = sign_authority(&view_private.into(), spki).into();
156
157                // Verify authority signature
158                subaddr
159                    .view_public_key()
160                    .verify_authority(spki, &RistrettoSignature::try_from(&sig[..]).unwrap())
161                    .unwrap();
162
163                // Check local impl matches upstream
164                assert_eq!(
165                    sig,
166                    subaddr.fog_authority_sig().unwrap(),
167                    "Fog authority signature mismatch"
168                );
169            }
170        }
171    }
